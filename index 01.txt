        .chat-box { height: 200px; overflow-y: scroll; border: 1px solid #ddd; padding: 10px; border-radius: 5px; background: #fafafa; }

<body>

    <h2>🏫 سامانه کلاس آنلاین</h2>
    
    <!-- ۱. بخش حضور و غیاب -->
    <div class="card">
        <h3>📌 حضور و غیاب</h3>
        <p id="user-info">در حال دریافت اطلاعات کاربر...</p>
        <button onclick="registerAttendance()">ثبت حضور در کلاس</button>
        <p id="attendance-status" style="color: green;"></p>
    </div>

    <!-- ۲. بخش چت آنلاین کلاس -->
    <div class="card">
        <h3>💬 چت کلاس</h3>
        <div id="chat-box" class="chat-box"></div>
        <input type="text" id="msg-input" placeholder="پیام خود را بنویسید...">
        <button style="background-color: #007bff;" onclick="sendMessage()">ارسال پیام</button>
    </div>

    <script>
        // ۱. مقداردهی اولیه Supabase (اطلاعات خود را اینجا جایگزین کنید)
        const SUPABASE_URL = 'https://cxnztsdhnndijpyscvvg.supabase.co';
        const SUPABASE_ANON_KEY = '
sb_publishable_6df-9vl0Iraj7Ou0SgnV3g_1_xDneAm';
        const supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        // ۲. دریافت شناسه کاربر (از URL ایتا یا حالت تست)
        const urlParams = new URLSearchParams(window.location.search);
        // اگر شناسه در URL نبود، یک آیدی تست ایجاد می‌کند
        const studentId = urlParams.get('user_id') || 1001; 
        const studentName = urlParams.get('name') || "دانش‌آموز نمونه";

        document.getElementById('user-info').innerText = `خوش آمدید: ${studentName} (شناسه: ${studentId})`;

        // ثبت یا آپدیت دانش‌آموز در دیتابیس
        async function initUser() {
            await supabase.from('students').upsert({ id: studentId, full_name: studentName });
            loadMessages();
            listenToNewMessages();
        }

        // ۳. عملکرد ثبت حضور و غیاب
        async function registerAttendance() {
            const { data, error } = await supabase.from('attendance').insert([
                { student_id: studentId }
            ]);
            
            if (error) {
                alert("خطا در ثبت حضور: " + error.message);
            } else {
                document.getElementById('attendance-status').innerText = "✅ حضور شما با موفقیت ثبت شد!";
            }
        }

        // ۴. ارسال پیام در چت
        async function sendMessage() {
            const input = document.getElementById('msg-input');
            const content = input.value.trim();
            if (!content) return;

            await supabase.from('messages').insert([
                { student_id: studentId, sender_name: studentName, content: content }
            ]);
            input.value = '';
        }

        // ۵. دریافت پیام‌های چت
        async function loadMessages() {
            const { data: messages } = await supabase.from('messages').select('*').order('created_at', { ascending: true });
            const chatBox = document.getElementById('chat-box');
            chatBox.innerHTML = '';
            messages.forEach(msg => {
                chatBox.innerHTML += `<div class="chat-msg"><b>${msg.sender_name}:</b> ${msg.content}</div>`;
            });
            chatBox.scrollTop = chatBox.scrollHeight;
        }

        // ۶. دریافت لحظه‌ای پیام‌های جدید (Realtime Chat)
        function listenToNewMessages() {
            supabase
                .channel('public:messages')
                .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'messages' }, payload => {
                    const msg = payload.new;
                    const chatBox = document.getElementById('chat-box');
                    chatBox.innerHTML += `<div class="chat-msg"><b>${msg.sender_name}:</b> ${msg.content}</div>`;
                    chatBox.scrollTop = chatBox.scrollHeight;
                })
                .subscribe();
        }

        // اجرا هنگام ورود به صفحه
        initUser();
    </script>
